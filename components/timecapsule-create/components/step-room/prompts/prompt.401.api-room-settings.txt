```typescript
import axios from 'axios';
import { OrderResponse, RoomSettingsResponse } from '../types';

// .env에 이미 설정된 API URL 사용
const API_BASE_URL = process.env.EXPO_PUBLIC_API_BASE_URL || 'http://localhost:3000';

/**
 * 1단계) 주문 정보 조회 API
 * @param orderId 주문 ID (UUID)
 * @returns 주문 정보 (snake_case), order.capsule_id를 추출하여 2단계에서 사용
 * @throws 404: 존재하지 않는 orderId
 * @throws 500: 서버 내부 오류
 */
export async function getOrderInfo(orderId: string): Promise<OrderResponse> {
  try {
    const response = await axios.get<OrderResponse>(
      `${API_BASE_URL}/api/orders/${orderId}`,
      {
        headers: {
          'Content-Type': 'application/json',
        },
      }
    );

    return response.data;
  } catch (error) {
    if (axios.isAxiosError(error)) {
      if (error.response?.status === 404) {
        throw new Error('주문 정보를 찾을 수 없습니다.');
      }
      throw new Error(`Order API 호출 실패: ${error.response?.status || 'Network Error'}`);
    }
    console.error('주문 정보 조회 실패:', error);
    throw error;
  }
}

/**
 * 2단계) 대기실 설정값 조회 API
 * @param capsuleId 캡슐 ID (UUID) - 1단계에서 추출한 order.capsule_id
 * @returns 대기실 설정값 (snake_case)
 * @throws 404: 존재하지 않는 capsuleId 또는 주문 정보 없음
 * @throws 500: 서버 내부 오류
 */
export async function getRoomSettings(capsuleId: string): Promise<RoomSettingsResponse> {
  try {
    const response = await axios.get<RoomSettingsResponse>(
      `${API_BASE_URL}/api/capsules/step-rooms/${capsuleId}/settings`,
      {
        headers: {
          'Content-Type': 'application/json',
        },
      }
    );

    return response.data;
  } catch (error) {
    if (axios.isAxiosError(error)) {
      if (error.response?.status === 404) {
        throw new Error('대기실 또는 주문 정보를 찾을 수 없습니다.');
      }
      throw new Error(`Room Settings API 호출 실패: ${error.response?.status || 'Network Error'}`);
    }
    console.error('대기실 설정값 조회 실패:', error);
    throw error;
  }
}
```

```typescript
import { useState, useEffect } from 'react';
import { getOrderInfo, getRoomSettings } from '../api/capsule';
import { OrderResponse, RoomSettingsResponse } from '../types';
import { mockRoomData } from '../constants';

export function useRoomData(orderId: string) {
  const [orderInfo, setOrderInfo] = useState<OrderResponse | null>(null);
  const [roomSettings, setRoomSettings] = useState<RoomSettingsResponse | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    async function loadRoomData() {
      setIsLoading(true);
      setError(null);

      try {
        // ⭐ 1단계: Order 조회 → capsule_id 추출
        const orderData = await getOrderInfo(orderId);
        setOrderInfo(orderData);

        const capsuleId = orderData.order.capsule_id;

        // capsule_id가 null인 경우 처리
        if (!capsuleId) {
          throw new Error('주문에 연결된 캡슐 ID가 없습니다.');
        }

        // ⭐ 2단계: capsule_id로 Room Settings 조회
        const roomData = await getRoomSettings(capsuleId);
        setRoomSettings(roomData);

        // RoomData 변환 (내부 변환만, snake_case → camelCase)
        setRoomData({
          capsuleId: roomData.room_id,
          capsuleName: roomData.capsule_name,
          openDate: roomData.open_date,
          maxParticipants: roomData.max_participants,
          // ...
        });
      } catch (err) {
        console.warn('API 호출 실패, 목데이터 사용:', err);
        setError(err instanceof Error ? err : new Error('Unknown error'));

        // 목데이터로 폴백 (snake_case)
        setRoomSettings(mockRoomData);
      } finally {
        setIsLoading(false);
      }
    }

    if (orderId) {
      loadRoomData();
    }
  }, [orderId]);

  return {
    orderInfo,
    roomSettings,
    isLoading,
    error,
  };
}
```

  title={roomData?.capsuleName || '제목 없음'}
  openDate={roomData?.openDate || '날짜 미정'}
  maxParticipants={roomData?.maxParticipants || 1}

{roomData?.additionalOptions.hasMusicFile && <VoiceRecordButton />}
{roomData?.additionalOptions.hasVideo && <VideoUploadButton />}
{Array.from({ length: roomData?.imageSlots || 3 }).map((_, i) => (

에러처리가이드) 에러별 대응 전략

1. 404 Not Found (존재하지 않는 orderId) - 1단계 에러
   - 원인: 잘못된 orderId, 삭제된 주문

   - 대응: 목데이터 폴백 + 에러 토스트
   - 메시지: "결제 정보를 확인 중입니다. 잠시 후 다시 시도해주세요."

   - 원인: 잘못된 초대 링크, 삭제된 캡슐
   - 대응: 목데이터 폴백 + 에러 토스트
   - 메시지: "대기실 정보를 찾을 수 없습니다."

4. 404 Not Found (주문 정보 없음) - 2단계 에러
   - 원인: 결제 미완료 캡슐, 데이터 불일치
   - 대응: 목데이터 폴백 + 에러 토스트
   - 메시지: "캡슐 정보를 불러올 수 없습니다."

   - 원인: 서버 장애, DB 오류
   - 대응: 목데이터 폴백 + 재시도 버튼 (선택사항)
   - 메시지: "일시적인 오류가 발생했습니다."

6. Network Error

   - 메시지: "네트워크 연결을 확인해주세요."

7. Timeout Error
   - 원인: API 응답 지연
   - 대응: 목데이터 폴백 + 재시도 버튼
   - 메시지: "요청 시간이 초과되었습니다."
